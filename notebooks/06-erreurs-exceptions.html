💡 Solution 
def convertir_prix(texte):
    try:
        return float(texte.replace(",", "."))
    except ValueError:
        print(f"Prix invalide : {texte!r}")
        return None

print(convertir_prix("19.9"))
print(convertir_prix("12,5"))
print(convertir_prix("gratuit"))

rejets = 0
for valeur in brut:
    try:

5. Plusieurs except

Complète delai_moyen pour qu’elle gère :

  • un fournisseur absent du dictionnaire (KeyError) : afficher Fournisseur inconnu : ... ;
  • un fournisseur sans livraison (ZeroDivisionError) : afficher Aucune livraison pour ....

Dans les deux cas, la fonction renvoie None.

delais = {
    "Acier SA": [3, 5, 4],
    "BoisPlus": [10, 12],
    "NouveauFournisseur": [],
}

def delai_moyen(fournisseur):
    # TODO : entoure ce calcul d'un try / except
    jours = delais[fournisseur]
    return sum(jours) / len(jours)

print(delai_moyen("Acier SA"))
print(delai_moyen("NouveauFournisseur"))
print(delai_moyen("Plastico"))

# Résultat attendu :
# 4.0
# Aucune livraison pour NouveauFournisseur
# None
# Fournisseur inconnu : Plastico
# None
💡 Solution 
def delai_moyen(fournisseur):
    try:
        jours = delais[fournisseur]
        return sum(jours) / len(jours)
    except KeyError:
        print(f"Fournisseur inconnu : {fournisseur}")
    except ZeroDivisionError:
        print(f"Aucune livraison pour {fournisseur}")
    return None

print(delai_moyen("Acier SA"))
print(delai_moyen("NouveauFournisseur"))
print(delai_moyen("Plastico"))

  • une ValueError si quantite est inférieure ou égale à 0.

# Résultat attendu :
# Commande : 50 x vis
# ValueError : La quantité doit être positive, reçu 0
# TypeError : La quantité doit être un entier, reçu str
# ValueError : La quantité doit être positive, reçu -2
💡 Solution 
def commander(article, quantite):
    """Prépare une ligne de commande.

    Args:
        article: Nom de l'article.
        quantite: Nombre d'unités commandées.

    Returns:
        La ligne de commande sous forme de texte.

    Raises:
        TypeError: Si la quantité n'est pas un entier.
        ValueError: Si la quantité est inférieure ou égale à 0.
    """
    if not isinstance(quantite, int):
        raise TypeError(f"La quantité doit être un entier, reçu {type(quantite).__name__}")
    if quantite <= 0:
        raise ValueError(f"La quantité doit être positive, reçu {quantite}")
    return f"Commande : {quantite} x {article}"

print(commander("vis", 50))

for q in [0, "10", -2]:
    try:
        commander("vis", q)
    except (TypeError, ValueError) as e:
        print(f"{type(e).__name__} : {e}")

💡 except (TypeError, ValueError) attrape les deux types avec un seul bloc, et type(e).__name__ donne le nom de l’erreur.

8. Saisie robuste

Écris une boucle qui redemande un nombre de palettes tant que la saisie n’est pas un entier strictement positif.

(Cellule interactive : sur Colab ou dans VS Code, une zone de saisie apparaît.)

# TODO

# Exemple d'exécution :
# Nombre de palettes : trois
# Ce n'est pas un nombre entier.
# Nombre de palettes : -1
# Le nombre doit être positif.
# Nombre de palettes : 3
# 3 palettes enregistrées
💡 Solution 
while True:
    saisie = input("Nombre de palettes : ")
    try:
        palettes = int(saisie)
    except ValueError:
        print("Ce n'est pas un nombre entier.")
        continue
    if palettes > 0:
        break
    print("Le nombre doit être positif.")

print(f"{palettes} palettes enregistrées")

9. Pour aller plus loin : tes propres exceptions

Crée deux exceptions métier :

  • EntrepotError, l’erreur de base ;
  • CapaciteDepasseeError, qui en hérite.

Puis écris ranger(entrepot, palettes) qui ajoute des palettes à entrepot["occupe"], et lève une CapaciteDepasseeError si cela dépasse entrepot["capacite"].

# TODO : les deux classes d'exceptions


def ranger(entrepot, palettes):
    # TODO
    pass


entrepot = {"nom": "Lyon", "capacite": 100, "occupe": 70}
for arrivage in [20, 15, 10]:
    try:
        ranger(entrepot, arrivage)
        print(f"{arrivage} palettes rangées ({entrepot['occupe']}/{entrepot['capacite']})")
    except CapaciteDepasseeError as e:
        print(f"⚠️ {e}")

# Résultat attendu :
# 20 palettes rangées (90/100)
# ⚠️ Lyon : impossible de ranger 15 palettes, il reste 10 places
# 10 palettes rangées (100/100)
💡 Solution 
class EntrepotError(Exception):
    """Erreur de base pour la gestion d'entrepôt."""


class CapaciteDepasseeError(EntrepotError):
    """L'entrepôt n'a pas assez de place."""


def ranger(entrepot, palettes):
    places_libres = entrepot["capacite"] - entrepot["occupe"]
    if palettes > places_libres:
        raise CapaciteDepasseeError(
            f"{entrepot['nom']} : impossible de ranger {palettes} palettes, "
            f"il reste {places_libres} places"
        )
    entrepot["occupe"] += palettes


entrepot = {"nom": "Lyon", "capacite": 100, "occupe": 70}
for arrivage in [20, 15, 10]:
    try:
        ranger(entrepot, arrivage)
        print(f"{arrivage} palettes rangées ({entrepot['occupe']}/{entrepot['capacite']})")
    except CapaciteDepasseeError as e:
        print(f"⚠️ {e}")

🤔 Pourquoi créer EntrepotError alors qu’on ne l’utilise pas directement ? Parce qu’un except EntrepotError attrapera toutes les erreurs de ton application (capacité, référence inconnue, etc.) sans toucher aux vrais bugs Python.
