Exercice 10

for n in range(1, 31):
    if n % 3 == 0 and n % 5 == 0:   # le cas le plus restrictif en premier !
        print("FizzBuzz")
    elif n % 3 == 0:
        print("Fizz")
    elif n % 5 == 0:
        print("Buzz")
    else:
        print(n)
