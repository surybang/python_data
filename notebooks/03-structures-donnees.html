Exercice 10

# 1
ca_total = 0
for v in ventes:
    ca_total += v["montant"]

# 2
villes = []
for v in ventes:
    villes.append(v["ville"])
villes = sorted(set(villes))

# 3
ca_par_ville = {}
for v in ventes:
    ville = v["ville"]
    ca_par_ville[ville] = ca_par_ville.get(ville, 0) + v["montant"]

# 4
produits_lyon = []
for v in ventes:
    if v["ville"] == "Lyon":
        produits_lyon.append(v["produit"])
Avec pandas, ces quatre questions s’écriront df["montant"].sum(), sorted(df["ville"].unique()), df.groupby("ville")["montant"].sum() et df[df["ville"] == "Lyon"]["produit"]. Vous venez de les coder à la main !
